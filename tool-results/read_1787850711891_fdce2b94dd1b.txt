   247→function detectFormat(payload) {
   248→  if (!payload || typeof payload !== 'object') return { format: null, confidence: 0 };
   249→
   250→  // ATC: has card_id + payload + signature
   251→  if (payload.card_id && payload.payload && payload.signature) {
   252→    return { format: 'atc-v2', confidence: 0.95 };
   253→  }
   254→
   255→  // EAT-AI: CBOR/CWT — detect by structure (Uint8Array or specific CWT claims)
   256→  if (payload instanceof Uint8Array || (payload.cwt && payload.cwt.length > 0)) {
   257→    return { format: 'eat-ai', confidence: 0.80 };
   258→  }
   259→
   260→  // ZTA: Anthropic format — has zta_version or agent_id + trust.score
   261→  if (payload.zta_version || (payload.agent_id && payload.trust && payload.trust.score !== undefined && !payload.card_id)) {
   262→    return { format: 'zta', confidence: 0.85 };
   263→  }
   264→
   265→  // A2A Agent Card: has capabilities + service_endpoint + version
   266→  if (payload.capabilities && (payload.service_endpoint || payload.url) && payload.version && !payload.signature) {
   267→    return { format: 'a2a-card', confidence: 0.80 };
   268→  }
   269→
   270→  // MCP Server Card: has protocolVersion + tools + serverInfo
   271→  if (payload.protocolVersion && payload.tools && payload.serverInfo) {
   272→    return { format: 'mcp-card', confidence: 0.90 };
   273→  }
   274→
   275→  return { format: null, confidence: 0 };
   276→}

   278→function toUTS(format, payload) {
   279→  switch (format) {
   280→    case 'atc-v2': return atcToUTS(payload);
   281→    case 'eat-ai': return eatToUTS(payload);
   282→    case 'zta': return ztaToUTS(payload);
   283→    case 'a2a-card': return a2aToUTS(payload);
   284→    case 'mcp-card': return mcpToUTS(payload);
   285→    default: return null;
   286→  }
   287→}
   288→
   289→function fromUTS(format, uts) {
   290→  switch (format) {
   291→    case 'atc-v2': return utsToATC(uts);
   292→    case 'eat-ai': return utsToEAT(uts);
   293→    case 'zta': return utsToZTA(uts);
   294→    case 'a2a-card': return utsToA2A(uts);
   295→    case 'mcp-card': return utsToMCP(uts);
   296→    default: return null;
   297→  }
   298→}
   299→
   300→function verifyByFormat(format, payload, caPublicKey) {
   301→  switch (format) {
   302→    case 'atc-v2': return verifyATC(payload, caPublicKey);
   303→    case 'eat-ai': return verifyEAT(payload, caPublicKey);
   304→    case 'zta': return verifyZTA(payload, caPublicKey);
   305→    case 'a2a-card': return verifyA2A(payload, caPublicKey);
   306→    case 'mcp-card': return verifyMCP(payload, caPublicKey);
   307→    default: return { valid: false, format, issues: ['No adapter for verification'], warnings: [] };
   308→  }
   309→}

   314→function atcToUTS(card) {
   315→  const p = card.payload || {};
   316→  const sig = card.signature || {};
   317→  const t = p.trust || {};
   318→  const id = p.identity || {};
   319→  const cap = p.capabilities || {};
   320→  const meta = p.metadata || {};
   321→  const warnings = [];
   322→
   323→  if (!sig.ca_key_id) warnings.push('v2_violation: ca_key_id missing');
   324→  if (!sig.evidence_hash) warnings.push('v2_violation: evidence_hash missing');
   325→  if (sig.canonical_json && sig.canonical_json !== 'RFC_8785_JCS') warnings.push(`Deprecated: ${sig.canonical_json}`);
   326→
   327→  const evidence = [];
   328→  for (const [layer, passed] of Object.entries(t.audit_layers_passed || {})) {
   329→    if (passed) evidence.push({ type: 'sentinel-audit', source: layer, result: 'pass', timestamp: meta.issued_at });
   330→  }
   331→
   332→  return {
   333→    uts_version: '1.0.0',
   334→    subject: { id: p.agent_id || card.card_id, name: p.agent_name || 'Unknown', type: 'agent' },
   335→    identity: { public_key: id.public_key, key_algorithm: id.key_algorithm || 'Ed25519', key_id: sig.ca_key_id },
   336→    trust: { score: t.sentinel_review_score || 0, confidence: t.risk_level === 'low' ? 'high' : 'medium', evidence, assessor: meta.issuer || 'MarketNow', assessed_at: meta.issued_at, expires_at: meta.expires_at },
   337→    capabilities: { provides: cap.provides || [], requires: [], protocols: [cap.protocol_language || 'mcp'] },
   338→    provenance: { source: 'marketnow', original_signature_hash: sig.evidence_hash, original_format: 'atc-v2' },
   339→    lifecycle: { issued_at: meta.issued_at, expires_at: meta.expires_at, revoked: card.status === 'revoked', version: p.schema_version || '2.0.0' },
   340→    format: { type: 'atc-v2', version: p.schema_version || '2.0.0', raw: card },
   341→    warnings,
   342→  };
   343→}

   345→function utsToATC(uts) {
   346→  const id = `ATC-${Date.now().toString(36).toUpperCase()}`;
   347→  return {
   348→    card_id: id, status: uts.lifecycle.revoked ? 'revoked' : 'active',
   349→    payload: {
   350→      card_id: id, schema_version: '2.0.0', decision_authority: 'consumer',
   351→      agent_id: uts.subject.id, agent_name: uts.subject.name,
   352→      identity: { public_key: uts.identity.public_key || 'MCowBQYDK2VwAyEA', key_algorithm: uts.identity.key_algorithm || 'Ed25519' },
   353→      trust: { sentinel_review_score: uts.trust.score, sentinel_score: uts.trust.score, audit_layers_passed: {}, composite_trust: uts.trust.score, risk_level: uts.trust.confidence === 'high' ? 'low' : 'medium' },
   354→      capabilities: { provides: uts.capabilities.provides, protocol_language: uts.capabilities.protocols[0] || 'mcp', translate: true },
   355→      payment: { method: 'none', wallet_address: null },
   356→      metadata: { issued_at: uts.lifecycle.issued_at, expires_at: uts.lifecycle.expires_at, issuer: uts.trust.assessor, revocation_url: `https://marketnow.site/api/atc?action=verify&card_id=${id}` },
   357→    },
   358→    signature: { algorithm: 'Ed25519 (RFC 8032)', value: '00'.repeat(64), signed_by: uts.trust.assessor, signed_at: uts.lifecycle.issued_at, canonical_json: 'RFC_8785_JCS', ca_key_id: uts.identity.key_id || 'MCowBQYDK2VwAyEA', evidence_hash: uts.provenance.original_signature_hash || 'sha256:pending', policy_version: uts.lifecycle.version || '2.0.0' },
   359→  };
   360→}

   362→function verifyATC(card, caKey) {
   363→  const issues = [], warnings = [];
   364→  if (!card?.payload || !card?.signature) return { valid: false, format: 'atc-v2', issues: ['missing payload/signature'], warnings };
   365→  const sig = card.signature, p = card.payload;
   366→  if (!sig.ca_key_id) warnings.push('v2_violation: ca_key_id missing');
   367→  if (!sig.evidence_hash) warnings.push('v2_violation: evidence_hash missing');
   368→  if (sig.canonical_json && sig.canonical_json !== 'RFC_8785_JCS') issues.push(`v2: ${sig.canonical_json} deprecated`);
   369→  if (p.metadata?.expires_at && new Date(p.metadata.expires_at) < new Date()) issues.push('expired');
   370→  if (card.status === 'revoked') issues.push('revoked');
   371→  return { valid: issues.length === 0, format: 'atc-v2', uts: atcToUTS(card), issues, warnings, v2_compliant: !warnings.some(w => w.startsWith('v2_violation')) };
   372→}

   377→function eatToUTS(payload) {
   378→  // EAT-AI uses CWT (CBOR Web Token) — for simplicity we accept JSON-decoded claims
   379→  const claims = typeof payload === 'string' ? JSON.parse(payload) : payload;
   380→  const warnings = [];
   381→
   382→  // EAT-specific: UEID (Universal Entity ID), OTE (Originating Trusted Environment)
   383→  if (claims.ueid) warnings.push('EAT UEID present — TEE attestation may not translate to all formats');
   384→  if (claims.ote) warnings.push('EAT OTE (Trusted Environment) present — hardware attestation');
   385→
   386→  return {
   387→    uts_version: '1.0.0',
   388→    subject: { id: claims.sub || 'unknown', name: claims.name || claims.sub || 'EAT Entity', type: 'agent' },
   389→    identity: { public_key: claims.cnf?.jwk, key_algorithm: 'ES256', key_id: claims.kid },
   390→    trust: {
   391→      score: claims.trust_score || 0,
   392→      confidence: claims.trust_level === 'high' ? 'high' : claims.trust_level === 'medium' ? 'medium' : 'low',
   393→      evidence: claims.evidence || [],
   394→      assessor: claims.iss || 'IETF EAT Issuer',
   395→      assessed_at: claims.iat ? new Date(claims.iat * 1000).toISOString() : undefined,
   396→      expires_at: claims.exp ? new Date(claims.exp * 1000).toISOString() : undefined,
   397→    },
   398→    capabilities: { provides: claims.capabilities || [], requires: [], protocols: ['cwt'] },
   399→    provenance: { source: 'ietf-eat', original_format: 'eat-ai' },
   400→    lifecycle: {
   401→      issued_at: claims.iat ? new Date(claims.iat * 1000).toISOString() : undefined,
   402→      expires_at: claims.exp ? new Date(claims.exp * 1000).toISOString() : undefined,
   403→      revoked: false, version: 'draft-00',
   404→    },
   405→    format: { type: 'eat-ai', version: 'draft-00', raw: claims },
   406→    warnings,
   407→  };
   408→}

   410→function utsToEAT(uts) {
   411→  // Convert UTS to EAT-AI claims (JSON representation — CBOR encoding would be next step)
   412→  return {
   413→    iss: uts.trust.assessor,
   414→    sub: uts.subject.id,
   415→    name: uts.subject.name,
   416→    iat: Math.floor(new Date(uts.lifecycle.issued_at).getTime() / 1000),
   417→    exp: uts.lifecycle.expires_at ? Math.floor(new Date(uts.lifecycle.expires_at).getTime() / 1000) : undefined,
   418→    trust_score: uts.trust.score,
   419→    trust_level: uts.trust.confidence,
   420→    evidence: uts.trust.evidence,
   421→    capabilities: uts.capabilities.provides,
   422→    cnf: uts.identity.public_key ? { jwk: uts.identity.public_key } : undefined,
   423→    ueid: uts.identity.attestation?.quote,
   424→    note: 'JSON representation of CWT claims — encode to CBOR for wire format',
   425→  };
   426→}
   427→
   428→function verifyEAT(payload, caKey) {
   429→  const uts = eatToUTS(payload);
   430→  const issues = [];
   431→  if (!uts.subject.id) issues.push('EAT: missing sub claim');
   432→  if (!uts.trust.assessor) issues.push('EAT: missing iss claim');
   433→  return { valid: issues.length === 0, format: 'eat-ai', uts, issues, warnings: uts.warnings || [] };
   434→}

   439→function ztaToUTS(payload) {
   440→  return {
   441→    uts_version: '1.0.0',
   442→    subject: { id: payload.agent_id || payload.id || 'unknown', name: payload.agent_name || payload.name || 'ZTA Agent', type: 'agent' },
   443→    identity: { public_key: payload.identity?.public_key, key_algorithm: payload.identity?.key_algorithm || 'Ed25519' },
   444→    trust: {
   445→      score: payload.trust?.score || 0,
   446→      confidence: payload.trust?.confidence || 'medium',
   447→      evidence: payload.trust?.evidence || [],
   448→      assessor: payload.trust?.assessor || 'Anthropic',
   449→      assessed_at: payload.metadata?.issued_at || payload.issued_at,
   450→    },

   558→// ============================================================================
   559→// PROOF-OF-POSSESSION (PoP) — Audit item #4
   560→// ============================================================================
   561→// Prevents credential theft: even if an attacker steals the ATC JSON,
   562→// they can't use it without the agent's private key.
   563→//
   564→// Flow:
   565→//   1. Caller requests a nonce: GET /api/trust?action=nonce&agent_id=X
   566→//   2. Agent signs the nonce with its private key
   567→//   3. Caller submits: POST /api/trust?action=verify with PoP
   568→//   4. UTA verifies: signature + nonce + agent_id match
   569→
   570→// In-memory nonce store (production: use Upstash Redis)
   571→const nonceStore = new Map();
   572→
   573→function generateNonce() {
   574→  const crypto = require('crypto');
   575→  return crypto.randomBytes(32).toString('hex');
   576→}
   577→
   578→function verifyPoP(agent_id, nonce, signature, publicKey) {
   579→  const crypto = require('crypto');
   580→  try {
   581→    // The agent signs: SHA256(agent_id + ":" + nonce)
   582→    const message = `${agent_id}:${nonce}`;
   583→    const messageBytes = Buffer.from(message, 'utf-8');
   584→    const sigBytes = Buffer.from(signature, 'hex');
   585→    return crypto.verify(null, messageBytes, publicKey, sigBytes);
   586→  } catch (e) {
   587→    return false;
   588→  }
   589→}

   593→// The agent must sign this nonce and submit it with the verify call

   595→// ── ARTIFACT BINDING — Audit item #5 ──
   596→// The ATC credential carries a cryptographic link to the source artifact:
   597→//   provenance.artifact_hash = sha256(git_commit_sha + npm_tarball_sha256 + docker_digest)
   598→// This prevents supply-chain attacks where the repo differs from the published package.
   599→
   600→function computeArtifactBinding(gitSha, npmTarballSha256, dockerDigest) {
   601→  const crypto = require('crypto');
   602→  const binding = JSON.stringify({
   603→    git_commit_sha: gitSha,
   604→    npm_tarball_sha256: npmTarballSha256,
   605→    docker_digest: dockerDigest,
   606→  });
   607→  return `sha256:${crypto.createHash('sha256').update(binding).digest('hex')}`;
   608→}

   610→// ── MUTATION TESTS — Audit item #6 ──
   611→// Run mutation tests: change 1 byte in the payload and verify the signature breaks
   612→async function runMutationTests(payload) {
   613→  const crypto = require('crypto');
   614→  const mutations = [
   615→    { field: 'trust.sentinel_review_score', change: (v) => v + 1 },
   616→    { field: 'trust.sentinel_score', change: (v) => v + 1 },
   617→    { field: 'agent_id', change: (v) => v + '_mutated' },
   618→    { field: 'agent_name', change: (v) => v + ' MUTATED' },
   619→    { field: 'identity.public_key', change: (v) => v.slice(0, -4) + 'XXXX' },
   620→    { field: 'capabilities.provides', change: (v) => [...v, 'admin'] },
   621→    { field: 'metadata.expires_at', change: (v) => '2099-12-31T23:59:59Z' },
   622→    { field: 'metadata.issuer', change: (v) => 'FAKE_ISSUER' },
   623→    { field: 'signature.ca_key_id', change: (v) => 'FAKE_KEY_ID' },
   624→    { field: 'signature.policy_version', change: (v) => '0.0.0' },
   625→  ];
   626→  
   627→  const results = [];
   628→  for (const mutation of mutations) {
   629→    const mutated = JSON.parse(JSON.stringify(payload));
   630→    const parts = mutation.field.split('.');
   631→    let obj = mutated;
   632→    for (let i = 0; i < parts.length - 1; i++) obj = obj[parts[i]];
   633→    const key = parts[parts.length - 1];
   634→    const original = obj[key];
   635→    obj[key] = mutation.change(original);
   636→    
   637→    // Verify: the mutation should break the signature
   638→    const result = verifyATC(mutated, null);
   639→    results.push({
   640→      mutation: mutation.field,
   641→      original: String(original).slice(0, 30),
   642→      mutated: String(obj[key]).slice(0, 30),
   643→      expected: 'invalid',
   644→      actual: result.valid ? 'VALID (BUG!)' : 'invalid (correct)',
   645→      passed: !result.valid,
   646→    });
   647→  }
   648→  return results;
   649→}
