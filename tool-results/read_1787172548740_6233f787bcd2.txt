     1→/**
     2→ * MarketNow — Agent Trust Card (ATC) API — REAL IMPLEMENTATION
     3→ * =============================================================
     4→ *
     5→ * SSL certificates for AI agents. MarketNow acts as the Certificate
     6→ * Authority (CA). Every ATC is:
     7→ *   1. Cryptographically SIGNED with the CA Ed25519 private key
     8→ *   2. PERSISTED to _data/atc/{card_id}.json via GitHub API (durable)
     9→ *   3. VERIFIABLE by anyone using the CA public key (GET /api/atc?action=ca-key)
    10→ *   4. REVOCABLE — revoked cards are marked in the persisted record
    11→ *
    12→ * Operations:
    13→ *   POST   /api/atc  {action:"issue", agent_id, public_key, ...}
    14→ *     → Signs the ATC payload with CA Ed25519 key, persists to GitHub,
    15→ *       returns the signed ATC.
    16→ *   GET    /api/atc?action=verify&card_id=X
    17→ *     → Fetches the ATC from GitHub, verifies the signature with CA
    18→ *       public key, checks expiry, checks revocation status.
    19→ *   POST   /api/atc  {action:"revoke", card_id, reason}
    20→ *     → Updates the persisted ATC record: status="revoked".
    21→ *       Future verify calls will return valid=false.
    22→ *   GET    /api/atc
    23→ *     → Lists all ATCs from _data/atc/ directory.
    24→ *   GET    /api/atc?action=ca-key
    25→ *     → Returns the CA public key (Ed25519 SPKI PEM).
    26→ *   POST   /api/atc  {action:"translate", from, to, message}
    27→ *     → Translates agent protocol messages (LangChain ↔ MCP ↔ AutoGen ↔ CrewAI).
    28→ *
    29→ * Sentinel score integration:
    30→ *   When issuing, if `skill_id` is provided, the ATC's sentinel_score
    31→ *   is fetched from the actual Sentinel certificate in _data/sentinel_certificates/.
    32→ *   If `repo_url` is provided, we run a real-time audit via /api/audit-skill.
    33→ *
    34→ * Cryptography:
    35→ *   - Algorithm: Ed25519 (RFC 8032)
    36→ *   - Private key: Vercel env var MARKETNOW_ATC_CA_PRIVATE_KEY (PKCS8 PEM)
    37→ *   - Public key: committed to _data/atc/ca-public-key.json (SPKI PEM)
    38→ *   - Signature: detached, over the canonical JSON of the ATC payload
    39→ */

    85→// ─── CA key loading ──────────────────────────────────────────────────────

    87→function loadCAKeys() {
    88→  if (_caPrivateKey) return { privateKey: _caPrivateKey, publicKey: _caPublicKey, publicKeyPem: _caPublicKeyPem };
    89→
    90→  if (!CA_PRIVATE_KEY_PEM) {
    91→    throw new Error('CA private key not configured. Set MARKETNOW_ATC_CA_PRIVATE_KEY env var.');
    92→  }
    93→
    94→  _caPrivateKey = crypto.createPrivateKey(CA_PRIVATE_KEY_PEM);
    95→  _caPublicKey = crypto.createPublicKey(_caPrivateKey);
    96→  _caPublicKeyPem = _caPublicKey.export({ type: 'spki', format: 'pem' }).trim();
    97→  return { privateKey: _caPrivateKey, publicKey: _caPublicKey, publicKeyPem: _caPublicKeyPem };
    98→}

   100→// ─── Signing & verification ─────────────────────────────────────────────
   101→
   102→/**
   103→ * Canonical JSON serialization using RFC 8785 (JCS).
   104→ * Replaces the ad-hoc recursive sort with the international standard.
   105→ * Fixes the canonicalization bug reported by @anp2network permanently.
   106→ */
   107→function canonicalJson(obj) {
   108→  return rfc8785Canonicalize(obj);
   109→}
   110→
   111→/**
   112→ * Sign data with Ed25519 CA private key.
   113→ * Returns signature as hex string.
   114→ */
   115→function signATC(payload) {
   116→  const { privateKey } = loadCAKeys();
   117→  const data = Buffer.from(canonicalJson(payload), 'utf8');
   118→  const signature = crypto.sign(null, data, privateKey); // null = Ed25519 has no algorithm param
   119→  return signature.toString('hex');
   120→}
   121→
   122→/**
   123→ * Verify an ATC signature.
   124→ * @param {Object} payload - the ATC payload (without signature field)
   125→ * @param {string} signatureHex - hex signature
   126→ * @returns {boolean}
   127→ */
   128→function verifySignature(payload, signatureHex) {
   129→  try {
   130→    const { publicKey } = loadCAKeys();
   131→    const data = Buffer.from(canonicalJson(payload), 'utf8');
   132→    const signature = Buffer.from(signatureHex, 'hex');
   133→    return crypto.verify(null, data, publicKey, signature);
   134→  } catch (e) {
   135→    return false;
   136→  }
   137→}

   141→async function ghApiCall(method, path, body) {
   142→  if (!GITHUB_TOKEN) throw new Error('GitHub token not configured');
   143→  const url = `https://api.github.com/repos/${REPO}/contents/${encodeURIComponent(path)}?ref=${encodeURIComponent(BRANCH)}`;
   144→  const headers = {
   145→    Authorization: `Bearer ${GITHUB_TOKEN}`,
   146→    Accept: 'application/vnd.github+json',
   147→    'User-Agent': 'marketnow-atc',
   148→  };
   149→  const opts = { method, headers };
   150→  if (body) {
   151→    opts.headers['Content-Type'] = 'application/json';
   152→    opts.body = JSON.stringify(body);
   153→  }
   154→  const r = await fetch(url, opts);
   155→  return r;
   156→}

   158→async function fetchATC(card_id, { skipCache = false } = {}) {
   159→  // Check cache (unless skipCache — verify always reads fresh)
   160→  if (!skipCache) {
   161→    const cached = _atcCache.get(card_id);
   162→    if (cached && Date.now() - cached.fetchedAt < ATC_CACHE_TTL_MS) {
   163→      return cached.data;
   164→    }
   165→  }
   166→
   167→  // ── WORKAROUND: GitHub account shadowbanned. Use static file bundled in
   168→  // this deployment first. Falls back to GitHub Contents API if not found.
   169→  // NOTE: Static file reflects the state at build time. For verify, we want
   170→  // fresh data, so we still try GitHub first (if token works) for accuracy,
   171→  // then fall back to static for the list endpoint.
   172→  // For verify, we use the static file because the GitHub API also returns
   173→  // 404 for shadowbanned accounts.
   174→  try {
   175→    const baseUrl = process.env.VERCEL_URL
   176→      ? `https://${process.env.VERCEL_URL}`
   177→      : 'https://marketnow.site';
   178→    const r = await fetch(`${baseUrl}/api/atc/${encodeURIComponent(card_id)}.json`);
   179→    if (r.ok) {
   180→      const data = await r.json();
   181→      _atcCache.set(card_id, { data, fetchedAt: Date.now() });
   182→      return data;
   183→    }
   184→  } catch (e) {
   185→    // Static file not available — fall through to GitHub API
   186→  }
   187→
   188→  // ── FALLBACK: GitHub Contents API (works for non-flagged accounts) ──
   189→  const url = `https://api.github.com/repos/${REPO}/contents/${ATC_DIR}/${encodeURIComponent(card_id)}.json?ref=${encodeURIComponent(BRANCH)}`;
   190→  try {
   191→    const r = await fetch(url, {
   192→      headers: {
   193→        Authorization: `Bearer ${GITHUB_TOKEN}`,
   194→        Accept: 'application/vnd.github+json',
   195→        'User-Agent': 'marketnow-atc',
   196→      },
   197→    });
   198→    if (r.status === 404) return null;
   199→    if (!r.ok) throw new Error(`GitHub ${r.status}`);
   200→    const meta = await r.json();
   201→    // Content is base64-encoded
   202→    const content = Buffer.from(meta.content, 'base64').toString('utf8');
   203→    const data = JSON.parse(content);
   204→    _atcCache.set(card_id, { data, fetchedAt: Date.now() });
   205→    return data;
   206→  } catch (e) {
   207→    return null;
   208→  }
   209→}

   211→async function listATCs() {
   212→  // ── OPTIMIZED: Read _index.json (1 API call) instead of 58 ──
   213→  // The _index.json file contains a summary of all ATCs (card_id, status,
   214→  // agent_id, score, etc.) — enough for the list endpoint.
   215→  // Falls back to the old method (58 calls) if _index.json doesn't exist.
   216→  
   217→  // Check cache first (5 minute TTL)
   218→  if (_atcIndexCache && (Date.now() - _atcIndexFetchedAt) < ATC_INDEX_CACHE_TTL_MS) {
   219→    return _atcIndexCache;
   220→  }
   221→
   222→  // Helper: convert index entries to the format expected by callers
   223→  function indexToAtcs(index) {
   224→    return (index.cards || []).map(c => ({
   225→      card_id: c.card_id,
   226→      status: c.status,
   227→      payload: {
   228→        card_id: c.card_id,
   229→        agent_id: c.agent_id,
   230→        agent_name: c.agent_name,
   231→        trust: {
   232→          sentinel_review_score: c.sentinel_review_score,
   233→          sentinel_score: c.sentinel_review_score, // backward compat
   234→          risk_level: c.risk_level,
   235→        },
   236→        metadata: {
   237→          issued_at: c.issued_at,
   238→          expires_at: c.expires_at,
   239→        },
   240→      },
   241→    }));
   242→  }
   243→
   244→  // ── WORKAROUND: GitHub account is shadowbanned (flagged as spam).
   245→  // raw.githubusercontent.com and api.github.com/repos return 404 even for
   246→  // public repos when the account is flagged. The static atc-index.json
   247→  // file is bundled into this deployment and served from /api/atc-index.json.
   248→  // This bypasses GitHub entirely for the list endpoint.
   249→  // See: https://support.github.com/contact to resolve the account flag.
   250→  try {
   251→    const staticUrl = `https://${process.env.VERCEL_URL ? '' : 'marketnow.site'}${process.env.VERCEL_URL || ''}/api/atc-index.json`;
   252→    const baseUrl = process.env.VERCEL_URL
   253→      ? `https://${process.env.VERCEL_URL}`
   254→      : 'https://marketnow.site';
   255→    const r = await fetch(`${baseUrl}/api/atc-index.json`);
   256→    if (r.ok) {
   257→      const index = await r.json();
   258→      const atcs = indexToAtcs(index);
   259→      _atcIndexCache = atcs;
   260→      _atcIndexFetchedAt = Date.now();
   261→      return atcs;
   262→    }
   263→  } catch (e) {
   264→    // Static file not available — fall through to GitHub methods
   265→  }
   266→
   267→  // ── FALLBACK 1: Try raw.githubusercontent with Bearer token ──
   268→  // (works for non-flagged accounts; fails for flagged accounts)
   269→  try {
   270→    const rawUrl = `https://raw.githubusercontent.com/${REPO}/${encodeURIComponent(BRANCH)}/${ATC_DIR}/_index.json`;
   271→    const r = await fetch(rawUrl, {
   272→      headers: {
   273→        Authorization: `Bearer ${GITHUB_TOKEN}`,
   274→        'User-Agent': 'marketnow-atc',
   275→      },
   276→    });
   277→    
   278→    if (r.ok) {
   279→      const text = await r.text();
   280→      const index = JSON.parse(text);
   281→      const atcs = indexToAtcs(index);
   282→      _atcIndexCache = atcs;
   283→      _atcIndexFetchedAt = Date.now();
   284→      return atcs;
   285→    }
   286→  } catch (e) {
   287→    // _index.json not found or parse error — fall through to old method
   288→  }
   289→  
   290→  // ── FALLBACK: Old method (58 API calls) if _index.json doesn't exist ──
   291→  const url = `https://api.github.com/repos/${REPO}/contents/${ATC_DIR}?ref=${encodeURIComponent(BRANCH)}`;
   292→  try {
   293→    const r = await fetch(url, {
   294→      headers: {
   295→        Authorization: `Bearer ${GITHUB_TOKEN}`,
   296→        Accept: 'application/vnd.github+json',
   297→        'User-Agent': 'marketnow-atc',
   298→      },
   299→    });
   300→    if (!r.ok) return [];
   301→    const files = await r.json();
   302→    if (!Array.isArray(files)) return [];
   303→    const atcFiles = files.filter(f => f.type === 'file' && f.name.startsWith('ATC-') && f.name.endsWith('.json'));
   304→
   305→    const atcs = [];
   306→    for (let i = 0; i < atcFiles.length; i += 5) {
   307→      const batch = atcFiles.slice(i, i + 5);
   308→      const results = await Promise.all(batch.map(async f => {
   309→        try {
   310→          const fileUrl = `https://api.github.com/repos/${REPO}/contents/${ATC_DIR}/${encodeURIComponent(f.name)}?ref=${encodeURIComponent(BRANCH)}`;
   311→          const fr = await fetch(fileUrl, {
   312→            headers: {
   313→              Authorization: `Bearer ${GITHUB_TOKEN}`,
   314→              Accept: 'application/vnd.github+json',
   315→              'User-Agent': 'marketnow-atc',
   316→            },
   317→          });
   318→          if (!fr.ok) return null;
   319→          const meta = await fr.json();
   320→          if (!meta.content) return null;
   321→          const content = Buffer.from(meta.content, 'base64').toString('utf8');
   322→          return JSON.parse(content);
   323→        } catch { return null; }
   324→      }));
   325→      atcs.push(...results.filter(Boolean));
   326→    }
   327→    
   328→    // Cache the result
   329→    _atcIndexCache = atcs;
   330→    _atcIndexFetchedAt = Date.now();
   331→    return atcs;
   332→  } catch (e) {
   333→    return [];
   334→  }
   335→}

   337→async function persistATC(card_id, atc) {
   338→  // Get the existing file's SHA (if updating) so GitHub can replace it
   339→  let sha = null;
   340→  try {
   341→    const metaUrl = `https://api.github.com/repos/${REPO}/contents/${ATC_DIR}/${encodeURIComponent(card_id)}.json?ref=${encodeURIComponent(BRANCH)}`;
   342→    const metaR = await fetch(metaUrl, {
   343→      headers: {
   344→        Authorization: `Bearer ${GITHUB_TOKEN}`,
   345→        Accept: 'application/vnd.github+json',
   346→        'User-Agent': 'marketnow-atc',
   347→      },
   348→    });
   349→    if (metaR.ok) {
   350→      const meta = await metaR.json();
   351→      sha = meta.sha;
   352→    }
   353→  } catch {}
   354→
   355→  const content = Buffer.from(JSON.stringify(atc, null, 2)).toString('base64');
   356→  const url = `https://api.github.com/repos/${REPO}/contents/${ATC_DIR}/${encodeURIComponent(card_id)}.json`;
   357→  const body = {
   358→    message: `${sha ? 'update' : 'issue'} ATC ${card_id}`,
   359→    content,
   360→    branch: BRANCH,
   361→  };
   362→  if (sha) body.sha = sha;
   363→
   364→  const r = await fetch(url, {
   365→    method: 'PUT',
   366→    headers: {
   367→      Authorization: `Bearer ${GITHUB_TOKEN}`,
   368→      Accept: 'application/vnd.github+json',
   369→      'User-Agent': 'marketnow-atc',
   370→      'Content-Type': 'application/json',
   371→    },
   372→    body: JSON.stringify(body),
   373→  });
   374→
   375→  if (!r.ok) {
   376→    const errBody = await r.text();
   377→    throw new Error(`GitHub persist failed: ${r.status} ${errBody.slice(0, 200)}`);
   378→  }
   379→
   380→  // Invalidate cache
   381→  _atcCache.delete(card_id);
   382→
   383→  // ── Trigger sync-atc-static workflow to update static files + redeploy Vercel ──
   384→  // The GitHub push above will trigger the sync-atc-static.yml workflow (via
   385→  // `paths: _data/atc/**`), which regenerates public/api/atc-index.json +
   386→  // public/api/atc/{card_id}.json and triggers a Vercel deploy.
   387→  // No explicit trigger needed here — GitHub Actions handles it automatically.
   388→  return true;
   389→}

   395→  const url = `https://raw.githubusercontent.com/${REPO}/${encodeURIComponent(BRANCH)}/_data/sentinel_certificates/${encodeURIComponent(skill_id)}.json`;

   412→  } catch {
   413→    return null;
   414→  }

   417→// ─── Handler ────────────────────────────────────────────────────────────
   418→
   419→export default async function handler(req, res) {
   420→  res.setHeader('Content-Type', 'application/json; charset=utf-8');
   421→  res.setHeader('Cache-Control', 'no-store, no-cache, must-revalidate');
   422→  setCorsHeaders(req, res);
   423→  applySecurityHeaders(res);
   424→  res.setHeader('Vary', 'Origin');
   425→
   426→  if (req.method === 'OPTIONS' || req.method === 'HEAD') return res.status(200).end();
   427→
   428→  // Derive action from URL path if not explicitly set.
   429→  // Vercel rewrites /api/submit-skill and /api/referrals to /api/atc,
   430→  // so we detect the original path here.
   431→  const urlPath = req.url?.split('?')[0] || req.path || '';
   432→  const bodyAction = (req.body || {}).action;
   433→  let action = req.query.action;
   434→
   435→  if (urlPath.endsWith('/submit-skill')) {
   436→    // POST /api/submit-skill → action=submit-skill
   437→    // GET  /api/submit-skill?submission_id=sub_xxx → action=submission-status
   438→    action = req.method === 'POST' ? 'submit-skill' : 'submission-status';
   439→  } else if (urlPath.endsWith('/referrals')) {
   440→    // Map /api/referrals body.action values to internal action names
   441→    if (req.method === 'POST') {
   442→      if (bodyAction === 'mint') action = 'mint-referral';
   443→      else if (bodyAction === 'credit') action = 'credit-referral';
   444→      else if (bodyAction === 'click') action = 'click-referral';
   445→      else action = 'referrals-help';
   446→    } else {
   447→      // GET /api/referrals?action=lookup&ref_code=xxx
   448→      if (req.query.action === 'lookup') action = 'referral-lookup';
   449→      else if (req.query.action === 'list') action = 'referral-list';
   450→      else action = 'referrals-help';
   451→    }
   452→  } else if (!action && bodyAction) {
   453→    action = bodyAction;
   454→  }
   455→
   456→  try {
   457→    // ─── GET handlers ──────────────────────────────────────────────────
   458→
   459→    if (req.method === 'GET') {
   460→      // ── trust: compact trust score for install decisions (legacy) ──
   461→      // The full Trust API is at POST /api/atc?action=trust (see below).
   462→      // This GET handler provides the legacy compact trust score lookup.
   463→      if (action === 'trust') {
   464→        const skillId = req.query?.skillId || req.query?.skill_id;
   465→        if (!skillId) {
   466→          return res.status(200).json({
   467→            service: 'MarketNow Trust API',
   468→            version: '2.0.0',
   469→            description: 'Unified trust decision endpoint — combines Sentinel, ATC, Policy, and Interceptor.',
   470→            endpoint: 'POST /api/atc?action=trust',
   471→            architecture: 'DISCOVER → SENTINEL → IDENTITY → TRUST → POLICY → ENFORCEMENT → AUDIT',
   472→            legacy_endpoint: 'GET /api/atc?action=trust&skillId=X (compact trust score only)',
   473→          });
   474→        }
   475→        try {
   476→          const baseUrl = process.env.VERCEL_URL ? `https://${process.env.VERCEL_URL}` : 'https://marketnow.site';
   477→          const resp = await fetch(`${baseUrl}/api/skills-lite.json`);
   478→          const skills = await resp.json();
   479→          const skill = skills.find(s => s.id === skillId || s.slug === skillId);
   480→          if (!skill) return res.status(404).json({ error: 'Skill not found', skillId });
   481→          const score = skill.sentinel_score || 0;
   482→          const risk = skill.risk_level || 'not_audited';
   483→          let recommendation;
   484→          if (score >= 8) recommendation = 'safe_to_install';
   485→          else if (score >= 5) recommendation = 'install_with_caution';
   486→          else recommendation = 'do_not_install';
   487→          return res.status(200).json({
   488→            skill_id: skill.id, skill_name: skill.name,
   489→            trust_score: score, max_score: 10, risk_level: risk,
   490→            recommendation,
   491→            certificate_url: `https://marketnow.site/api/audit-skill?certificate=1&skillId=${skill.id}`,
