    84→async function verifyUsdcTx(txHash, expectedAmountRaw, expectedFromWallet) {
    85→  // 1. Revisar cache de txHash (solo para resultados negativos — positivos se re-verifican con dedup)
    86→  const cachedResult = txCache.get(txHash);
    87→  if (cachedResult && !cachedResult.ok) {
    88→    return cachedResult;
    89→  }
    90→
    91→  // 2. Verificar via pool de RPCs
    92→  try {
    93→    const { result: receipt } = await baseRpc.call('eth_getTransactionReceipt', [txHash]);
    94→
    95→    if (!receipt) {
    96→      const result = { ok: false, code: 'tx_not_found', receipt: null };
    97→      txCache.set(txHash, result);
    98→      return result;
    99→    }
   100→    if (receipt.status !== '0x1') {
   101→      const result = { ok: false, code: 'tx_failed', receipt };
   102→      txCache.set(txHash, result);
   103→      return result;
   104→    }
   105→    if (!receipt.logs || !Array.isArray(receipt.logs)) {
   106→      const result = { ok: false, code: 'no_logs', receipt };
   107→      txCache.set(txHash, result);
   108→      return result;
   109→    }
   110→
   111→    for (const log of receipt.logs) {
   112→      if (log.address?.toLowerCase() !== USDC_CONTRACT.toLowerCase()) continue;
   113→      if (!log.topics || log.topics[0] !== TRANSFER_TOPIC) continue;
   114→      const from = '0x' + log.topics[1].slice(26);
   115→      const to = '0x' + log.topics[2].slice(26);
   116→      const value = BigInt(log.data);
   117→      if (to.toLowerCase() === PAYMENT_WALLET.toLowerCase()) {
   118→        // C3 FIX: amount exact match (con tolerancia de 1 wei por redondeo)
   119→        if (value !== BigInt(expectedAmountRaw)) {
   120→          const result = {
   121→            ok: false, code: 'amount_mismatch', receipt,
   122→            expected: expectedAmountRaw, received: Number(value),
   123→          };
   124→          txCache.set(txHash, result);
   125→          return result;
   126→        }
   127→        // C4 FIX: validar from wallet si se proporcionó
   128→        if (expectedFromWallet && from.toLowerCase() !== expectedFromWallet.toLowerCase()) {
   129→          const result = {
   130→            ok: false, code: 'wrong_sender', receipt,
   131→            expected_from: expectedFromWallet, actual_from: from,
   132→          };
   133→          txCache.set(txHash, result);
   134→          return result;
   135→        }
   136→        const result = { ok: true, from, to, amount: Number(value), receipt };
   137→        // NOTA: NO se cachea el resultado positivo aquí — el dedup check abajo
   138→        // determina si este txHash ya fue usado para emitir una licencia.
   139→        return result;
   140→      }
   141→    }
   142→    const result = { ok: false, code: 'no_transfer_to_marketnow', receipt };
   143→    txCache.set(txHash, result);
   144→    return result;
   145→  } catch (e) {
   146→    return { ok: false, code: 'rpc_error', error: e.message };
   147→  }
   148→}

   164→async function isTxHashUsed(txHash) {
   165→  const cfg = usedTxRepoConfig();
   166→  if (!cfg.token) return { used: false, error: 'no_token' };
   167→  const url = `https://raw.githubusercontent.com/${cfg.repo}/${encodeURIComponent(cfg.branch)}/${encodeURIComponent(cfg.path)}/${txHash.toLowerCase()}.json`;
   168→  try {
   169→    const r = await fetch(url, { headers: { Authorization: `Bearer ${GITHUB_TOKEN}`, 'User-Agent': 'marketnow-agent-purchase' } });
   170→    if (r.status === 200) {
   171→      const data = await r.json();
   172→      return { used: true, data };
   173→    }
   174→    if (r.status === 404) return { used: false };
   175→    return { used: false, error: `http_${r.status}` };
   176→  } catch (e) {
   177→    return { used: false, error: e.message };
   178→  }
   179→}

   189→    licenseKey,

   215→async function recordMandateSpend(req, mandateId, amount, txHash, skill) {
   216→  // FINDING P2 FIX: Direct module call (no HTTP fetch to /api/mandates).
   217→  // The old implementation POSTed to /api/mandates with _internal/_secret,
   218→  // traversing the public TLS edge. Now we call the shared module directly.
   219→  // C1 FIX preserved: throws on failure (fail-closed) — caller MUST abort.
   220→  try {
   221→    const result = await recordSpendDirect(mandateId, amount, txHash, skill);
   222→    mandateCache.invalidate(mandateId);
   223→    if (!result.ok && result.code !== 'already_recorded') {
   224→      throw new Error(`mandate spend rejected: ${result.code}`);
   225→    }
   226→    return result;
   227→  } catch (e) {
   228→    console.error('mandate spend FAILED (fail-closed):', e);
   229→    mandateCache.invalidate(mandateId);
   230→    throw e;  // FAIL CLOSED — caller MUST abort
   231→  }
   232→}

   237→

   250→          cancel: 'POST {action: "cancel", job_id: "job_xxx"} — cancel async job',
   251→        },
   252→      });
   253→    }
   254→
   255→    // Fetch job from GitHub ledger
   256→    const GITHUB_TOKEN = process.env.MANDATES_GITHUB_TOKEN;
   257→    const REPO = process.env.MANDATES_REPO || 'alicelabs-llc/marketnow';
   258→    const BRANCH = 'master';

   260→      const url = `https://api.github.com/repos/${REPO}/contents/_data/purchase_jobs/${encodeURIComponent(job_id)}.json?ref=${BRANCH}`;
   261→      const r = await fetch(url, {
   262→        headers: { Authorization: `Bearer ${GITHUB_TOKEN}`, Accept: 'application/vnd.github+json', 'User-Agent': 'marketnow-jobs' },
   263→      });
   264→      if (r.status === 404) {
   265→        return res.status(404).json({ error: 'job_not_found', job_id });
   266→      }
   267→      if (!r.ok) throw new Error(`GitHub ${r.status}`);
   268→      const meta = await r.json();

   270→      const job = JSON.parse(content);
   271→      return res.status(200).json({
   272→        job_id: job.job_id,
   273→        status: job.status,
   274→        skill_id: job.skill_id,
   275→        started_at: job.started_at,
   276→        completed_at: job.completed_at || null,
   277→        result: job.result || null,
   278→        error: job.error || null,
   279→        message: job.status === 'completed' ? 'Purchase completed successfully.'
   280→          : job.status === 'cancelled' ? 'Purchase was cancelled.'
   281→          : job.status === 'failed' ? `Purchase failed: ${job.error || 'unknown'}`
   282→          : 'Purchase in progress. Poll again in 10s.',
   283→      });
   284→    } catch (e) {
   285→      return res.status(500).json({ error: 'job_fetch_failed', message: e.message });
   286→    }
   287→  }
   288→
   289→  if (req.method !== 'POST') return res.status(405).json({ error: 'POST or GET only' });
   290→
   291→  // ============================================================
   292→  // CANCEL API: POST {action: "cancel", job_id: "..."}
   293→  // ============================================================
   294→  const body = req.body || {};
   295→  if (body.action === 'cancel') {
   296→    const { job_id } = body;
   297→    if (!job_id) return res.status(400).json({ error: 'job_id required for cancel' });
   298→
   299→    const GITHUB_TOKEN = process.env.MANDATES_GITHUB_TOKEN;
   300→    const REPO = process.env.MANDATES_REPO || 'alicelabs-llc/marketnow';
   301→    const BRANCH = 'master';
   302→
   303→    try {
   304→      // Fetch job
   305→      const url = `https://api.github.com/repos/${REPO}/contents/_data/purchase_jobs/${encodeURIComponent(job_id)}.json?ref=${BRANCH}`;
   306→      const metaR = await fetch(url, {
   307→        headers: { Authorization: `Bearer ${GITHUB_TOKEN}`, Accept: 'application/vnd.github+json', 'User-Agent': 'marketnow-jobs' },
   308→      });
   309→      if (metaR.status === 404) return res.status(404).json({ error: 'job_not_found', job_id });
   310→      if (!metaR.ok) throw new Error(`GitHub ${metaR.status}`);
   311→      const meta = await metaR.json();
   312→      const job = JSON.parse(Buffer.from(meta.content, 'base64').toString('utf8'));
   313→
   314→      if (job.status === 'completed') {
   315→        return res.status(200).json({
   316→          status: 'too_late',
   317→          job_id,
   318→          message: 'Purchase already completed. License was issued.',

   320→        });
   321→      }
   322→      if (job.status === 'cancelled') {
   323→        return res.status(200).json({ status: 'already_cancelled', job_id });

   325→

   328→      job.cancelled_at = new Date().toISOString();
   329→
   330→      const content = Buffer.from(JSON.stringify(job, null, 2)).toString('base64');

   332→        method: 'PUT',
   333→        headers: { Authorization: `Bearer ${GITHUB_TOKEN}`, Accept: 'application/vnd.github+json', 'User-Agent': 'marketnow-jobs', 'Content-Type': 'application/json' },
   334→        body: JSON.stringify({ message: `cancel job ${job_id}`, content, branch: BRANCH, sha: meta.sha }),

   336→
   337→      if (!putR.ok) throw new Error(`GitHub PUT ${putR.status}`);
   338→
   339→      return res.status(200).json({
   340→        status: 'cancelled',
   341→        job_id,
   342→        cancelled_at: job.cancelled_at,
   343→        message: 'Purchase cancelled. No license was issued. If USDC was sent, it will be refunded via reconciliation.',

   475→      }
   476→      const expired = mandate.expiresAt && new Date(mandate.expiresAt).getTime() < Date.now();
